\vspace{-10mm}

\begin{alertblock}{HIGHLIGHT}
\centering
\textbf{Finite response can erode a basin of attraction while the recovered state remains locally stable.}
\end{alertblock}

\begin{block}{Finite response changes the recovery dynamics}
\justifying

A recovery coordinate $\theta$ evolves on a landscape $U(\theta)$, while a responding state $x$ relaxes only at finite rate,
\begin{bigequation}
\dot\theta=\eta[g(\theta)-x],
\qquad
\dot x=\gamma(\theta-x),
\qquad
\rho=\frac{\eta}{\gamma}.
\end{bigequation}

Eliminating $x$ gives
\begin{bigequation}
\ddot\theta+
[1-\rho+\rho U''(\theta)]\dot\theta
+\rho U'(\theta)=0.
\end{bigequation}

\begin{center}
\includegraphics[width=0.72\linewidth]{figures/landscape.pdf}
\end{center}

\textbf{Finite response turns gradient recovery into an inertial system with state-dependent damping.}
The effective damping can change sign across a fixed non-convex landscape.
\end{block}

\begin{block}{The barrier sets the onset}
\justifying

Let $\theta_b$ denote the barrier saddle. The stable basin boundary changes side relative to the conservative separatrix when
\begin{bigequation}
\tcbhighmath[boxrule=3pt,colframe=boxbg,colback=white]{
\rho^*=
\frac{1}{g'(\theta_b)}
=
\frac{1}{1-U''(\theta_b)}
}
\end{bigequation}
whenever $g'$ reaches its maximum at the barrier.

For the single-barrier Helmholtz landscape,
\[
U(\theta)=\frac{\theta^2}{2}-\frac{\theta^3}{3},
\qquad
\rho^*=\frac12.
\]

\begin{center}
\includegraphics[width=0.88\linewidth]{figures/threshold.pdf}
\end{center}

\textbf{A local property of the barrier fixes the first loss of global basin volume.}
\end{block}

\begin{block}{A global measure of erosion}
\justifying

Let $\Omega_0$ be the conservative reference basin and $B_\rho$ the finite-response basin. The retained fraction is
\begin{bigequation}
C(\rho)=
\frac{\mu(B_\rho\cap\Omega_0)}
{\mu(\Omega_0)}.
\end{bigequation}

$C(\rho)=1$ means the reference perturbation set is fully retained; $C(\rho)<1$ measures the fraction lost through basin erosion.
\end{block}

\end{column}

\begin{block}{The basin moves before the fixed point destabilizes}

\begin{center}
\includegraphics[width=0.98\linewidth]{figures/basin_geometry.pdf}
\end{center}

The dashed curve is the instantaneous-response boundary; the solid curve is the stable manifold of the barrier at finite response. For $\rho>\rho^*$ it moves into the reference basin and removes recoverable perturbations.

\begin{center}
\textbf{The attractor is still locally stable, but its nonlinear resilience is already smaller.}
\end{center}

\end{block}

\begin{block}{The mechanism extends beyond one landscape}
\justifying

The same finite-response structure appears exactly in bistable FitzHugh--Nagumo activator--recovery dynamics. With
\begin{bigequation}
\rho=\frac{\alpha}{b^2\beta},
\qquad
\rho^*=\frac{1}{b}.
\end{bigequation}

\begin{center}
\includegraphics[width=0.90\linewidth]{figures/fhn.pdf}
\end{center}

The basin boundary erodes while the outer equilibria remain locally stable.
\end{block}

\begin{block}{Take-away}
\begin{itemize}
    \item Finite response generates state-dependent dissipation without external time-dependent forcing.
    \item Basin erosion can precede any local loss of stability.
    \item In the barrier-controlled case, local barrier geometry predicts the exact erosion threshold.
\end{itemize}
\end{block}
